// --- ★ランダム取得（Open Library版） --- async function huntBooksByHiragana(targetCount) { let candidates = []; let retry = 0; while (candidates.length < targetCount && retry < 10) { retry++; const q = HIRAGANA[Math.floor(Math.random() * HIRAGANA.length)]; log(`文字「${q}」で検索中...`); const page = Math.floor(Math.random() * 5) + 1; // ★ Open Library API const url = `https://openlibrary.org/search.json?title=${encodeURIComponent(q)}&limit=40&page=${page}`; try { const res = await fetch(url); const data = await res.json(); if (data.docs) { let tempIsbns = []; for (const doc of data.docs) { if (doc.isbn) { // ISBN13で日本の本（9784）を抽出 const isbn13 = doc.isbn.find(i => i.length === 13 && i.startsWith('9784') ); if (isbn13) { tempIsbns.push(isbn13); } } } // 重複除去 const uniqueIsbns = [...new Set(tempIsbns)]; if (uniqueIsbns.length > 0) { const verified = await fetchOpenBDBulk(uniqueIsbns); for (const book of verified) { if (book && book.summary && book.summary.title) { if (!candidates.find(c => c.summary.isbn === book.summary.isbn)) { candidates.push(book); } } if (candidates.length >= targetCount) break; } } else { log(`「${q}」→ ISBNヒットなし`); } } else { log(`「${q}」→ 検索ヒットなし`); } } catch (e) { console.error(e); log(`エラー: ${e.message}`); } } return candidates.slice(0, targetCount); }
